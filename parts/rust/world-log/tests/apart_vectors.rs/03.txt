#[test]
fn keeping_public_refuses_what_the_log_does_not_commit_to() {
    let (mut log, mut salt, said) = said_room();
    let made = mark(&mut log, 5, &said, "text", "asked", &mut salt).unwrap();
    let pair = log.take_out(&made).unwrap();
    let entries = log.entries().to_vec();
    let contents = log.contents().clone();
    let its_salt = pair[0].as_str().unwrap();
    let value = pair[1].as_str().unwrap();
    let other_salt = contents[&said]["time"][0].clone();
    let place = apart::Place {
        entry: Some(said.clone()),
        field: Some("text".to_owned()),
    };
    let ground = appeal();
    let decision = trial_decision(1);
    for (bad, expected) in [
        (
            json!([its_salt, format!("{value}!")]),
            Refused::Fingerprint(place.clone()),
        ),
        (
            json!([other_salt, value]),
            Refused::Fingerprint(place.clone()),
        ),
        (
            json!([its_salt.to_uppercase(), value]),
            Refused::Salt(place.clone()),
        ),
        (json!([7, value]), Refused::Salt(place.clone())),
        (json!([its_salt, 1.5]), Refused::Value(place.clone())),
        // What is not a salt and a value at all.
        (json!([its_salt]), Refused::Contents(place.clone())),
        (
            json!([its_salt, value, value]),
            Refused::Contents(place.clone()),
        ),
        (json!({"salt": its_salt}), Refused::Contents(place.clone())),
        (Value::Null, Refused::Contents(place.clone())),
    ] {
        assert_eq!(
            log.keep_public(
                6,
                &made,
                &bad,
                decided(&ground, &decision),
                &Map::new(),
                &mut salt
            ),
            Err(expected),
            "{bad}"
        );
        assert_eq!(log.entries(), entries.as_slice(), "{bad}");
        assert_eq!(log.contents(), &contents, "{bad}");
    }
}

#[test]
fn it_stays_redacted_only_once_it_is_out_of_the_contents() {
    let (mut log, mut salt, said) = said_room();
    let made = mark(&mut log, 5, &said, "text", "asked", &mut salt).unwrap();
    let before = log.entries().len();
    assert_eq!(
        stays(&mut log, 6, &made, "decided", 1, &mut salt),
        Err(Refused::StillThere)
    );
    assert_eq!(log.entries().len(), before);
    assert_eq!(
        redactions_of(&log, &said, "text")[0].state,
        RedactionState::Open
    );
    log.take_out(&made).unwrap();
    stays(&mut log, 6, &made, "decided", 1, &mut salt).unwrap();
    assert_eq!(
        redactions_of(&log, &said, "text")[0].state,
        RedactionState::Stands
    );
}

    let (mut log, mut salt, said) = said_room();

    }
}

#[test]
fn a_decision_alone_never_makes_a_finished_redaction_missing() {
    // The fault carried from part two: one entry of kind
    // field_kept_public turned a redacted field into a missing one. The
    // writer now puts the field back with it, or writes nothing.
    let (mut log, mut salt, said) = said_room();
    let made = mark(&mut log, 5, &said, "text", "asked", &mut salt).unwrap();
    let pair = log.take_out(&made).unwrap();
    let ground = appeal();
    let decision = trial_decision(1);
    log.keep_public(
        6,
        &made,
        &pair,
        decided(&ground, &decision),
        &Map::new(),
        &mut salt,
    )
    .unwrap();
    let (entries, contents) = log.into_parts();
    let states = check(&Value::Array(entries), &Value::Object(contents)).unwrap();
    assert!(
        states
            .values()
            .flat_map(|fields| fields.values())
            .all(|state| *state != State::Missing)
    );
}

#[test]
fn a_published_log_can_still_be_at_fault() {
    // The check can't make a publisher honest: it says what it finds.
    let (mut log, mut salt, said) = said_room();
    let made = mark(&mut log, 5, &said, "text", "asked", &mut salt).unwrap();
    keep(&mut log, 6, &made, "decided", 1, &mut salt).unwrap();
    let mut contents = log.contents().clone();
    contents[&said].as_object_mut().unwrap().remove("text");
    let entries = Value::Array(log.entries().to_vec());
    assert_eq!(
        check(&entries, &Value::Object(contents)).unwrap()[&said]["text"],
        State::Missing
    );
    // And a field left in the contents after the decision that it stays
    // redacted is unfinished, and never shown.
    let other = mark(&mut log, 7, &said, "time", "asked", &mut salt).unwrap();
    let left = Value::Object(log.contents().clone());
    log.take_out(&other).unwrap();
    stays(&mut log, 8, &other, "decided", 2, &mut salt).unwrap();
    let entries = Value::Array(log.entries().to_vec());
    assert_eq!(
        check(&entries, &left).unwrap()[&said]["time"],
        State::Unfinished
    );
    let joined = join(&entries, &left).unwrap();
    assert!(joined[3]["data"].get("time").is_none());
}

fn redactions_need_no_contents_and_join_gives_them() {
    let (mut log, mut salt, said) = said_room();
    let made = mark(&mut log, 5, &said, "text", "asked", &mut salt).unwrap();
    log.take_out(&made).unwrap();
    stays(&mut log, 6, &made, "decided", 1, &mut salt).unwrap();
    let (entries, contents) = log.into_parts();
    let entries = Value::Array(entries);
    let found = redactions_json(&redactions(&entries).unwrap());
    assert_eq!(
        found,
        json!({&said: {"text": [
            {"id": made, "short": made[..SHORT], "state": "stands"}]}})
    );
    let joined = join(&entries, &Value::Object(contents)).unwrap();
    assert_eq!(joined[3]["redactions"], found[&said]);
    assert_eq!(joined[1]["redactions"], json!({}));
}

#[test]
fn a_redactions_id_is_shown_with_the_incidents_ids() {
    // Short ids are worked out over incidents and redactions together,
    // so no two that a room shows begin alike.
    let (mut log, mut salt, said) = said_room();
    let made = mark(&mut log, 5, &said, "text", "asked", &mut salt).unwrap();
    let other = log.entries()[4]["hash"].as_str().unwrap().to_owned();
    let removal = log
        .append(
            9,
            REMOVED,
            None,
            &map(&json!({"entry": other, "office": "trial moderator",
                         "ground": {"kind": "law", "ref": "trial room law, part 3"},
                         "decision": trial_decision(1)})),
            &Map::new(),
            &mut salt,
        )
        .unwrap();
    let entries = Value::Array(log.entries().to_vec());
    let together = short_ids(&[removal.clone(), made.clone()]);
    assert_eq!(redactions_of(&log, &said, "text")[0].short, together[&made]);
    assert_eq!(
        incidents(&entries).unwrap()[&other].incidents[0].short,
        together[&removal]
    );
    // Two ids that begin alike are each shown one digit past where they
    // part.
    let alike = [
        format!("{}{}", &made[..20], "0".repeat(44)),
        format!("{}{}", &made[..20], "1".repeat(44)),
    ];
    assert!(short_ids(&alike).values().all(|short| short.len() == 21));
}

// A writer opened on a log and contents that were written before
// (`Log::open`). The same tests as `TestOpened` in tests/test_apart.py.

/// How many salts the trial room has drawn: one for each field kept
/// apart.
fn drawn(v: &Value) -> usize {
    cases(v, "written")
        .iter()
        .map(|step| step["apart"].as_object().unwrap().len())
        .sum()
}

fn opened(log: &Log) -> Log {
    Log::open(
        &Value::Array(log.entries().to_vec()),
        &Value::Object(log.contents().clone()),
    )
    .unwrap()
}

#[test]
fn it_opens_on_every_log_that_holds() {
    let v = vectors();
    for case in cases(&v, "holds") {
        let name = &case["name"];
        let log = Log::open(&case["log"], &case["contents"]).unwrap();
        assert_eq!(Value::Array(log.entries().to_vec()), case["log"], "{name}");
        assert_eq!(
            json!(log.head()),
            case["log"].as_array().unwrap().last().unwrap()["hash"],
            "{name}"
        );
        // As the writer keeps them: no empty map for an entry.
        let kept: Map<String, Value> = map(&case["contents"])
            .into_iter()
            .filter(|(_, fields)| !fields.as_object().unwrap().is_empty())
            .collect();
        assert_eq!(log.contents(), &kept, "{name}");
        let (entries, contents) = log.into_parts();
        let states = check(&Value::Array(entries), &Value::Object(contents)).unwrap();
        assert_eq!(states_json(&states), case["states"], "{name}");
    }
}

#[test]
fn it_refuses_what_the_check_refuses_for_the_same_reason() {
    let v = vectors();
    for case in cases(&v, "refused") {
        let name = &case["name"];
        let error = Log::open(&case["log"], &case["contents"])
            .err()
            .unwrap_or_else(|| panic!("{name}: opened"));
        assert_eq!(error.reason(), case["reason"], "{name}");
        assert_eq!(place(&error), case["where"], "{name}");
    }
    let good = &v["holds"][0]["log"];
    for odd in [json!(null), json!({}), json!("log"), json!(7)] {
        assert!(Log::open(&odd, &json!({})).is_err(), "{odd}");
        if odd != json!({}) {
            assert!(Log::open(good, &odd).is_err(), "{odd}");
        }
    }
}

fn it_can_be_closed_and_opened_at_every_step() {
    // The same log, however often the writer is closed and opened
    // between two acts.
    let v = vectors();
    let never_closed = room(&v, &mut trial_salts());
    let mut salt = trial_salts();
    let mut log = Log::new(&v["table"], &map(&v["first"]), "world_created", &mut salt).unwrap();
    for step in cases(&v, "written") {
        log = opened(&log);
        log.append(
            step["tick"].as_u64().unwrap(),
            step["kind"].as_str().unwrap(),
            step["agent"].as_str(),
            &map(&step["data"]),
            &map(&step["apart"]),
            &mut salt,
        )
        .unwrap();
    }
    assert_eq!(log.entries(), never_closed.entries());
    assert_eq!(log.contents(), never_closed.contents());
}

#[test]
fn a_salt_given_back_is_remembered_from_then_on() {
    // Found by the independent review. A field is taken out, the writer
    // is closed and opened, and the field is put back with its salt
    // given. An opened writer did not know that salt. It is told it, and
    // from then on refuses a source that repeats it, as the writer that
    // never closed does.
    let v = vectors();
    let (ground, decision) = (appeal(), trial_decision(1));
    let by = Decided {
        office: KEEPER,
        ground: &ground,
        decision: &decision,
        agent: None,
    };
    for reopen in [false, true] {
        let mut salt = trial_salts();
        let mut log = room(&v, &mut salt);
        let said = log.entries()[3]["hash"].as_str().unwrap().to_owned();
        let pair = log.contents()[&said]["text"].clone();
        let its_salt = pair[0].as_str().unwrap().to_owned();
        let remembered = world_log::sha256_hex(its_salt.as_bytes());
        let a_mark = mark(&mut log, 5, &said, "text", "asked", &mut salt).unwrap();
        log.take_out(&a_mark).unwrap();
        if reopen {
            log = opened(&log);
            assert!(!format!("{log:?}").contains(&remembered));
        }
        let before = (log.entries().to_vec(), log.contents().clone());
        let mut again = || its_salt.clone();
        // The entry that keeps it public draws that very salt for a
        // field of its own: refused, and nothing changed.
        let refused = log.keep_public(6, &a_mark, &pair, by, &why("kept"), &mut again);
        assert!(
            matches!(refused, Err(Refused::SaltTwice(_))),
            "{reopen}: {refused:?}"
        );
        assert_eq!(log.entries(), before.0.as_slice());
        assert_eq!(log.contents(), &before.1);
        // The field is still out. A writer that learned the salt only
        // for this forgets it again; one that drew it keeps remembering
        // it.
        assert_eq!(format!("{log:?}").contains(&remembered), !reopen);
        // With a fresh salt the field is kept public.
        log.keep_public(6, &a_mark, &pair, by, &why("kept"), &mut salt)
            .unwrap();
        assert_eq!(log.contents()[&said]["text"], pair);
        assert!(format!("{log:?}").contains(&remembered));
        // And a later entry that draws the field's salt again is
        // refused, so what the writer holds still checks.
        let refused = log.append(
            9,
            "said",
            None,
            &Map::new(),
            &map(&json!({"text": "again"})),
            &mut again,
        );
        assert!(
            matches!(refused, Err(Refused::SaltTwice(_))),
            "{reopen}: {refused:?}"
        );
        apart::check(
            &Value::Array(log.entries().to_vec()),
            &Value::Object(log.contents().clone()),
        )
        .unwrap();
    }
}

#[test]
fn a_salt_that_is_in_the_contents_is_not_used_again() {
    let v = vectors();
    let written = room(&v, &mut trial_salts());
    let mut log = opened(&written);
    // A source that starts again from the first trial salt.
    let mut again = trial_salts();
    let kept = map(&json!({"text": "again"}));
    assert_eq!(
        log.append(9, "said", None, &Map::new(), &kept, &mut again)
            .unwrap_err()
            .reason(),
        "a salt appears twice"
    );
    assert_eq!(log.entries(), written.entries());
    // And it takes the next one that is new.
    let mut new = trial_salts_from(drawn(&v));
    log.append(9, "said", None, &Map::new(), &kept, &mut new)
        .unwrap();
    assert_eq!(log.entries().len(), written.entries().len() + 1);
}

#[test]
fn a_field_left_in_after_the_decision_can_be_taken_out() {
    // Only a writer opened on a published log meets this: a mark that
    // stands, and its field still in the contents.
    let v = vectors();
    let case = &v["holds"][3];
    let mut log = Log::open(&case["log"], &case["contents"]).unwrap();
    let said = case["log"][3]["hash"].as_str().unwrap();
    assert_eq!(case["states"][said]["key"], "unfinished");
    let found = redactions_of(&log, said, "key");
    assert_eq!(found.len(), 1);
    assert_eq!(found[0].state, RedactionState::Stands);
    let pair = log.contents()[said]["key"].clone();
    assert_eq!(log.take_out(&found[0].id).unwrap(), pair);
    assert_eq!(field_state(&log, said, "key"), State::Redacted);
    assert!(!format!("{log:?}").contains(pair[0].as_str().unwrap()));
    assert_eq!(
        Value::Array(log.entries().to_vec()),
        case["log"],
        "the log is not touched"
    );
    // An open mark whose field is still there is finished the same way,
    // or answered.
    let other = case["log"][4]["hash"].as_str().unwrap();
    assert_eq!(case["states"][other]["text"], "unfinished");
    let open_one = redactions_of(&log, other, "text");
    assert_eq!(open_one[0].state, RedactionState::Open);
    let mut salt = trial_salts_from(1000);
    assert_eq!(
        stays(&mut log, 20, &open_one[0].id, "decided", 7, &mut salt),
        Err(Refused::StillThere)
    );
    log.take_out(&open_one[0].id).unwrap();
    stays(&mut log, 20, &open_one[0].id, "decided", 7, &mut salt).unwrap();
    assert_eq!(
        redactions_of(&log, other, "text")[0].state,
        RedactionState::Stands
    );
}

#[test]
fn the_table_is_read_from_the_log() {
    let v = vectors();
    let written = room(&v, &mut trial_salts());
    let mut log = opened(&written);
    let mut salt = trial_salts_from(drawn(&v));
    let wrong = map(&json!({"name": "not a field of said"}));
    assert_eq!(
        log.append(9, "said", None, &Map::new(), &wrong, &mut salt)
            .unwrap_err()
            .reason(),
        "a field is kept apart that the table does not name"
    );
    let open = map(&json!({"text": "in the open"}));
    assert_eq!(
        log.append(9, "said", None, &open, &Map::new(), &mut salt)
            .unwrap_err()
            .reason(),
        "a field the table keeps apart is in the open"
    );
}
