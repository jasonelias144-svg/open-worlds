    use chat_world::hash::body_for_signing;
    use chat_world::record::*;
    use chat_world::{Proof, ProofOrArray};

    fn sign_single(rec: &Record, signer: &Keypair) -> ProofOrArray {
        let body = body_for_signing(rec).unwrap();
        let sig = signer.sign(&body);
        ProofOrArray::Single(Proof::create(signer, &sig))
    }

    fn sign_dual(rec: &Record, subj: &Keypair, room_kp: &Keypair)
        -> ProofOrArray {
        let body = body_for_signing(rec).unwrap();
        let s1 = subj.sign(&body);
        let s2 = room_kp.sign(&body);
        ProofOrArray::array(vec![
            Proof::create(subj, &s1),
            Proof::create(room_kp, &s2),
        ])
    }

Four facts about the crate as it stands:

  - Room::create takes the room's keypair by value. room.keypair()
    gives it back by reference, so a test that makes a room can sign
    as the room.
  - A room made by RoomTemplate::chat signs every record itself as
    well (head_interval is 1). A joining there carries two proofs:
    the one who joins, and the room.
  - Join::new and Leave::new, in src/record.rs, make the bare record
    with a placeholder for its proof. A record's seq is its place in
    the log, and its prev is the hash of the record before it.
  - Room::join does not let a key that is already in the room join
    again: it writes a refusal into the log in place of a joining.
    So two joinings by one key can only be written by forging the
    second.

A test returns chat_world::error::Result<()>, as your others do.
