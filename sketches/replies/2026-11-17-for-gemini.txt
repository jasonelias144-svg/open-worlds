use crate::error::{Error, Result};
use serde::Serialize;
use serde_json::Value;
use sha2::{Digest, Sha256};

/// RFC 8785 JCS canonical form. Integers only, ASCII keys — a
/// constrained subset of JCS that our records always satisfy.
pub fn canonicalize<T: Serialize>(value: &T) -> Result<Vec<u8>> {
    let v = serde_json::to_value(value)
        .map_err(|e| Error::Serialization(e.to_string()))?;
    let mut out = String::new();
    walk(&v, &mut out);
    Ok(out.into_bytes())
}

/// Canonicalize with the `proof` field stripped. Used for signing.
pub fn body_for_signing<T: Serialize>(value: &T) -> Result<Vec<u8>> {
    let mut v = serde_json::to_value(value)
        .map_err(|e| Error::Serialization(e.to_string()))?;
    if let Value::Object(map) = &mut v {
        map.remove("proof");
    }
    let mut out = String::new();
    walk(&v, &mut out);
    Ok(out.into_bytes())
}

fn walk(v: &Value, out: &mut String) {
    match v {
        Value::Null => out.push_str("null"),
        Value::Bool(b) => out.push_str(if *b { "true" } else { "false" }),
        Value::Number(n) => {
            if let Some(i) = n.as_i64() {
                out.push_str(&i.to_string());
            } else if let Some(u) = n.as_u64() {
                out.push_str(&u.to_string());
            } else {
                out.push('0');
            }
        }
        Value::String(s) => write_string(s, out),
        Value::Array(arr) => {
            out.push('[');
            for (i, item) in arr.iter().enumerate() {
                if i > 0 { out.push(','); }
                walk(item, out);
            }
            out.push(']');
        }
        Value::Object(map) => {
            let mut keys: Vec<&String> = map.keys().collect();
            keys.sort();
            out.push('{');
            for (i, key) in keys.iter().enumerate() {
                if i > 0 { out.push(','); }
                write_string(key, out);
                out.push(':');
                walk(&map[*key], out);
            }
            out.push('}');
        }
    }
}

fn write_string(s: &str, out: &mut String) {
    out.push('"');
    for c in s.chars() {
        match c {
            '"' => out.push_str("\\\""),
            '\\' => out.push_str("\\\\"),
            '\u{08}' => out.push_str("\\b"),
            '\u{09}' => out.push_str("\\t"),
            '\u{0A}' => out.push_str("\\n"),
            '\u{0C}' => out.push_str("\\f"),
            '\u{0D}' => out.push_str("\\r"),
            c if (c as u32) < 0x20 => {
                out.push_str(&format!("\\u{:04x}", c as u32));
            }
            c => out.push(c),
        }
    }
    out.push('"');
}

pub fn sha256_hex(bytes: &[u8]) -> String {
    let mut h = Sha256::new();
    h.update(bytes);
    hex::encode(h.finalize())
}

/// Hash of the full record, including proofs. This is what `prev`,
/// `target_hash`, and `home_ref` reference.
pub fn hash_of<T: Serialize>(value: &T) -> Result<String> {
    Ok(sha256_hex(&canonicalize(value)?))
}

use crate::error::{Error, Result};
use ed25519_dalek::{Signature, Signer, SigningKey, Verifier, VerifyingKey};
use serde::{Deserialize, Deserializer, Serialize, Serializer};

#[derive(Clone, PartialEq, Eq, Hash, Debug)]
pub struct DidKey(String);

impl DidKey {
    pub fn from_verifying_key(vk: &VerifyingKey) -> Self {
        DidKey(format!("did:key:{}", hex::encode(vk.to_bytes())))
    }

    pub fn parse(s: &str) -> Result<Self> {
        let rest = s
            .strip_prefix("did:key:")
            .ok_or_else(|| Error::InvalidDid(format!("missing prefix: {s}")))?;
        if rest.len() != 64 {
            return Err(Error::InvalidDid(format!("wrong length: {s}")));
        }
        hex::decode(rest).map_err(|_| Error::InvalidDid(format!("bad hex: {s}")))?;
        Ok(DidKey(s.to_string()))
    }

    pub fn verifying_key(&self) -> Result<VerifyingKey> {
        let rest = self.0.strip_prefix("did:key:").unwrap();
        let bytes = hex::decode(rest).map_err(|_| Error::InvalidDid("bad hex".into()))?;
        let arr: [u8; 32] = bytes
            .try_into()
            .map_err(|_| Error::InvalidDid("wrong length".into()))?;
        VerifyingKey::from_bytes(&arr).map_err(|e| Error::InvalidDid(e.to_string()))
    }

    pub fn as_str(&self) -> &str {
        &self.0
    }
}

impl std::fmt::Display for DidKey {
    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
        f.write_str(&self.0)
    }
}

impl Serialize for DidKey {
    fn serialize<S: Serializer>(&self, s: S) -> std::result::Result<S::Ok, S::Error> {
        s.serialize_str(&self.0)
    }
}

impl<'de> Deserialize<'de> for DidKey {
    fn deserialize<D: Deserializer<'de>>(d: D) -> std::result::Result<Self, D::Error> {
        let s = String::deserialize(d)?;
        DidKey::parse(&s).map_err(serde::de::Error::custom)
    }
}

#[derive(Clone)]
pub struct Keypair {
    signing: SigningKey,
}

impl Keypair {
    pub fn generate() -> Self {
        use rand::rngs::OsRng;
        Keypair { signing: SigningKey::generate(&mut OsRng) }
    }

    pub fn did(&self) -> DidKey {
        DidKey::from_verifying_key(&self.signing.verifying_key())
    }

    pub fn sign(&self, msg: &[u8]) -> Signature {
        self.signing.sign(msg)
    }

    pub fn verify(&self, msg: &[u8], sig: &Signature) -> Result<()> {
        self.signing
            .verifying_key()
            .verify(msg, sig)
            .map_err(|e| Error::InvalidProof(e.to_string()))
    }
}

            .strip_prefix('z')
            .ok_or_else(|| Error::InvalidProof("missing z prefix".into()))?;
        let bytes = bs58::decode(encoded)
            .into_vec()
            .map_err(|e| Error::InvalidProof(e.to_string()))?;
        let arr: [u8; 64] = bytes
            .try_into()
            .map_err(|_| Error::InvalidProof("wrong sig length".into()))?;
        let sig = Signature::from_bytes(&arr);
        self.verification_method
            .verifying_key()?
            .verify_strict(body, &sig)
            .map_err(|e| Error::InvalidProof(e.to_string()))
    }
}

#[derive(Clone, Debug, Serialize, Deserialize)]
#[serde(untagged)]
pub enum ProofOrArray {
    Single(Proof),
    Array(Vec<Proof>),
}

impl ProofOrArray {
    pub fn single(p: Proof) -> Self { ProofOrArray::Single(p) }
    pub fn array(v: Vec<Proof>) -> Self { ProofOrArray::Array(v) }
    pub fn placeholder() -> Self { ProofOrArray::Single(Proof::placeholder()) }
    pub fn to_vec(&self) -> Vec<&Proof> {
        match self {
            ProofOrArray::Single(p) => vec![p],
            ProofOrArray::Array(v) => v.iter().collect(),
        }
    }
    pub fn len(&self) -> usize {
        match self {
            ProofOrArray::Single(_) => 1,
            ProofOrArray::Array(v) => v.len(),
        }
    }
    pub fn find_by_method(&self, did: &DidKey) -> Option<&Proof> {
        self.to_vec().into_iter().find(|p| &p.verification_method == did)
    }
}

use crate::did::{DidKey, Keypair};
use crate::error::{Error, Result};
use crate::head::WorldHead;
use crate::record::*;

use serde::{Deserialize, Serialize};

fn short_hash(h: &str) -> String {
    h.chars().take(16).collect()
}

#[derive(Clone, Debug, Serialize, Deserialize)]
pub struct RoomLog {
    pub world_id: DidKey,
    pub template: RoomTemplate,
    pub records: Vec<Record>,
    pub heads: Vec<WorldHead>,
}

#[derive(Clone, Debug)]
pub struct Member {
    pub tier: String,
    pub is_ai: bool,
}

pub struct Room {
    world_id: DidKey,
    template: RoomTemplate,
    keypair: Keypair,
    records: Vec<Record>,
    members: HashMap<DidKey, Member>,
    heads: Vec<WorldHead>,
}

impl Room {
    pub fn create(
        template: RoomTemplate,
        room_kp: Keypair,
        creator_kp: &Keypair,
    ) -> Result<Self> {
        let world_id = room_kp.did();
        let creator_id = creator_kp.did();
        let mut members = HashMap::new();
        members.insert(
            creator_id.clone(),
            Member { tier: "terminal".into(), is_ai: false },
        );
        let mut room = Room {
            world_id: world_id.clone(),
            template: template.clone(),
            keypair: room_kp,
            records: Vec::new(),
            members,
            heads: Vec::new(),
        };
        let rec = RoomCreated::new(world_id, template, creator_id);
        let signed = crate::sign_record_dual(
            rec, creator_kp, &room.keypair, Record::RoomCreated, |r, p| r.proof = p,
        )?;
        room.records.push(Record::RoomCreated(signed));
        Ok(room)
    }

    pub fn log(&self) -> RoomLog {
        let mut heads = self.heads.clone();
        if self.template.head_interval > 1 && self.records.len() > 1 {
            let len = self.records.len() as u64;
            let covered = heads.last().map(|h| h.length == len).unwrap_or(false);
            if !covered {
                if let Ok(tail_head) = self.emit_head() {
                    heads.push(tail_head);
                }
            }
        }
        RoomLog {
            world_id: self.world_id.clone(),
            template: self.template.clone(),
            records: self.records.clone(),
            heads,
        }
    }

    pub fn records(&self) -> &[Record] { &self.records }
    pub fn heads(&self) -> &[WorldHead] { &self.heads }
    pub fn last_hash(&self) -> Option<String> { self.records.last().and_then(|r| r.hash().ok()) }
    pub fn world_id(&self) -> &DidKey { &self.world_id }
    pub fn template(&self) -> &RoomTemplate { &self.template }
    pub fn members(&self) -> &HashMap<DidKey, Member> { &self.members }
    pub fn keypair(&self) -> &Keypair { &self.keypair }
    pub fn head_interval(&self) -> u64 { self.template.head_interval }

    pub fn is_redacted(&self, message_hash: &str) -> bool {
        let mut redact_hashes: Vec<String> = Vec::new();
        let mut resolved: HashSet<String> = HashSet::new();
        for rec in &self.records {
            match rec {
                Record::Redact(r) if r.target_hash == message_hash => {
                    if let Ok(h) = rec.hash() { redact_hashes.push(h); }
                }
                Record::HumanOverride(o) if o.action == "restore" => {
                    resolved.insert(o.overrides_hash.clone());
                }
                _ => {}
            }
        }
        redact_hashes.iter().any(|h| !resolved.contains(h))
    }

    fn next_seq(&self) -> u64 { self.records.len() as u64 }

    fn should_emit_head(&self) -> bool {
        let n = self.template.head_interval;
        if n <= 1 { return false; }
        let appends = self.records.len().saturating_sub(1) as u64;
        appends > 0 && appends % n == 0
    }

    pub(crate) fn push(&mut self, rec: Record) -> Result<()> {
        self.records.push(rec);
        if self.should_emit_head() {
            let head = self.emit_head()?;
            self.heads.push(head);
        }
        Ok(())
    }

    fn refuse(&mut self, requester: DidKey, request_hash: String, reason: &str) -> Result<Record> {
        let rec = Refusal::new(
            self.world_id.clone(), self.next_seq(), self.last_hash(),
            request_hash, requester, reason.into(),
        );
        let signed = crate::sign_record(rec, &self.keypair, Record::Refusal, |r, p| r.proof = p)?;
        let out = Record::Refusal(signed);
        self.push(out.clone())?;
        Ok(out)
    }

    fn rate_ok(&self, author: &DidKey) -> bool {
        let window = self.template.rate_limit_window as usize;
        let cap = self.template.rate_limit_count as usize;
        let recent = self.records.iter().rev().take(window)
            .filter(|r| matches!(r, Record::Message(m) if &m.author_id == author))
            .count();
        recent < cap
    }

    fn inter_ai_pace_ok(&self, is_ai: bool) -> bool {
        if !is_ai || self.template.inter_ai_pace == 0 {
            return true;
        }
        let mut last_ai_idx: Option<usize> = None;
        for (i, rec) in self.records.iter().enumerate().rev() {
            if let Record::Message(m) = rec {
                let ai = self.members.get(&m.author_id).map(|x| x.is_ai).unwrap_or(false);
                if ai {
                    last_ai_idx = Some(i);
                    break;
                }
            }
        }
        match last_ai_idx {
            None => true,
            Some(idx) => {
                let since = (self.records.len() - idx - 1) as u64;
                since >= self.template.inter_ai_pace
            }
        }
    }

    pub fn join(&mut self, subject_kp: &Keypair, tier: &str, is_ai: bool) -> Result<Record> {
        let subject = subject_kp.did();
        if self.members.contains_key(&subject) {
            return self.refuse(subject, "".into(), "already_member");
        }
        if tier_rank(tier).is_none() {
            return self.refuse(subject, "".into(), "unknown_tier");
        }
        if self.template.personal {
            return self.refuse(subject, "".into(), "personal_room");
        }
        if self.template.membership == "human_only" && is_ai {
            return self.refuse(subject, "".into(), "human_only_room");
        }
        if self.template.membership == "ai_only" && !is_ai {
            return self.refuse(subject, "".into(), "ai_only_room");
        }
        if self.template.join_policy == "invite" {
            return self.refuse(subject, "".into(), "invite_required");
        }
        if self.template.join_policy == "approval" {
            return self.refuse(subject, "".into(), "approval_required");
        }
        let rec = Join::new(
            self.world_id.clone(), self.next_seq(), self.last_hash(),
            subject.clone(), tier.into(), is_ai,
        );
        let signed = if self.template.head_interval <= 1 {
            crate::sign_record_dual(rec, subject_kp, &self.keypair, Record::Join, |r, p| r.proof = p)?
        } else {
            crate::sign_record(rec, subject_kp, Record::Join, |r, p| r.proof = p)?
        };
        self.members.insert(subject, Member { tier: tier.into(), is_ai });
        let out = Record::Join(signed);
        self.push(out.clone())?;
        Ok(out)
    }

    pub fn leave(&mut self, subject_kp: &Keypair) -> Result<Option<Record>> {
        let subject = subject_kp.did();
        if !self.members.contains_key(&subject) {
            return Ok(None);
        }
        let rec = Leave::new(
            self.world_id.clone(), self.next_seq(), self.last_hash(), subject.clone(),
        );
        let signed = crate::sign_record(rec, subject_kp, Record::Leave, |r, p| r.proof = p)?;
        self.members.remove(&subject);
        let out = Record::Leave(signed);
        self.push(out.clone())?;
        Ok(Some(out))
    }

    pub fn post(&mut self, author_kp: &Keypair, text: &str, kind: &str) -> Result<Record> {
        let author = author_kp.did();
        let Some(m) = self.members.get(&author).cloned() else {
            return self.refuse(author, "".into(), "not_a_member");
        };
        if tier_rank(&m.tier).unwrap_or(0) < tier_rank("communicate").unwrap() {
            return self.refuse(author, "".into(), "insufficient_tier");
        }
        if text.len() as u64 > self.template.max_msg_len {
            return self.refuse(author, "".into(), "message_too_long");
        }
        if !self.rate_ok(&author) {
            return self.refuse(author, "".into(), "rate_limit");
        }
        if !self.inter_ai_pace_ok(m.is_ai) {
            return self.refuse(author, "".into(), "inter_ai_pace");
        }
        let rec = Message::new(
            self.world_id.clone(), self.next_seq(), self.last_hash(),
            author.clone(), text.into(), kind.into(),
        );
        let signed = if self.template.head_interval <= 1 {
            crate::sign_record_dual(rec, author_kp, &self.keypair, Record::Message, |r, p| r.proof = p)?
        } else {
            crate::sign_record(rec, author_kp, Record::Message, |r, p| r.proof = p)?
        };
        let out = Record::Message(signed);
        self.push(out.clone())?;
        Ok(out)
    }

    pub fn redact(
        &mut self,
        subject_kp: &Keypair,
        target_hash: &str,
        rule: &str,
        reason: &str,
    ) -> Result<Record> {
        let subject = subject_kp.did();
        let Some(m) = self.members.get(&subject).cloned() else {
            return self.refuse(subject, "".into(), "not_a_member");
        };
        let target = self.records.iter()
            .find(|r| r.hash().ok().as_deref() == Some(target_hash));
        let target_msg = match target {
            Some(Record::Message(msg)) => msg,
            _ => return self.refuse(subject, "".into(), "target_not_found"),
        };
        if self.template.moderation == "none" {
            return self.refuse(subject, "".into(), "moderation_disabled");
        }
        let is_author = target_msg.author_id == subject;
        let is_mod = self.template.moderation == "moderator"
            && tier_rank(&m.tier).unwrap_or(0) >= tier_rank("control").unwrap();
        let actor_role = if is_author {
            "author"
        } else if is_mod {
            "moderator"
        } else {
            return self.refuse(subject, "".into(), "not_allowed");
        };
        if !self.template.redaction_rules.iter().any(|r| r == rule) {
            return self.refuse(subject, "".into(), "rule_not_allowed");
        }
        let rec = Redact::new(
            self.world_id.clone(), self.next_seq(), self.last_hash(),
            subject.clone(), target_hash.into(),
            actor_role.into(), rule.into(), reason.into(),
        );
        let signed = if self.template.head_interval <= 1 {
            crate::sign_record_dual(rec, subject_kp, &self.keypair, Record::Redact, |r, p| r.proof = p)?
        } else {
            crate::sign_record(rec, subject_kp, Record::Redact, |r, p| r.proof = p)?
        };
        let out = Record::Redact(signed);
        self.push(out.clone())?;
        Ok(out)
    }

    pub fn override_ai(&mut self, overrider_kp: &Keypair, overrides_hash: &str, reason: &str) -> Result<Record> {
        let overrider = overrider_kp.did();
        let Some(m) = self.members.get(&overrider).cloned() else {
            return Err(Error::Policy("overrider is not a member".into()));
        };
        if m.is_ai {
            return Err(Error::Policy("only humans can override".into()));
        }
        if tier_rank(&m.tier).unwrap_or(0) < tier_rank("control").unwrap() {
            return Err(Error::Policy("insufficient tier".into()));
        }
        let target = self.records.iter()
            .find(|r| r.hash().ok().as_deref() == Some(overrides_hash));
        let Some(target_rec) = target else {
            return Err(Error::Policy("target not found".into()));
        };
        let Some(author) = target_rec.author_id() else {
            return Err(Error::Policy("target has no author".into()));
        };
        let is_ai = self.members.get(author).map(|m| m.is_ai).unwrap_or(false);
        if !is_ai {
            return Err(Error::Policy("can only override AI records".into()));
        }
        if let Some(spec) = &self.template.moderator {
            if &spec.moderator_id == author && !spec.can_be_overridden {
                return Err(Error::Policy("this AI cannot be overridden".into()));
            }
        }
        let rec = HumanOverride::new(
            self.world_id.clone(), self.next_seq(), self.last_hash(),
            overrides_hash.into(), overrider.clone(), "none".into(), reason.into(),
        );
        let signed = if self.template.head_interval <= 1 {
            crate::sign_record_dual(rec, overrider_kp, &self.keypair, Record::HumanOverride, |r, p| r.proof = p)?
        } else {
            crate::sign_record(rec, overrider_kp, Record::HumanOverride, |r, p| r.proof = p)?
        };
        let out = Record::HumanOverride(signed);
        self.push(out.clone())?;
        Ok(out)
    }

    pub fn restore_incident(&mut self, restorer_kp: &Keypair, redact_hash: &str, reason: &str) -> Result<Record> {
        let restorer = restorer_kp.did();
        let Some(m) = self.members.get(&restorer).cloned() else {
            return Err(Error::Policy("restorer is not a member".into()));
        };
        if m.is_ai {
            return Err(Error::Policy("only humans can restore".into()));
        }
        if tier_rank(&m.tier).unwrap_or(0) < tier_rank("control").unwrap() {
            return Err(Error::Policy("insufficient tier".into()));
        }
        let target = self.records.iter()
            .find(|r| r.hash().ok().as_deref() == Some(redact_hash));
        match target {
            Some(Record::Redact(_)) => {}
            _ => return Err(Error::Policy("restore target must be a redact record".into())),
        }
        let rec = HumanOverride::new(
            self.world_id.clone(), self.next_seq(), self.last_hash(),
            redact_hash.into(), restorer.clone(), "restore".into(), reason.into(),
        );
        let signed = if self.template.head_interval <= 1 {
            crate::sign_record_dual(rec, restorer_kp, &self.keypair, Record::HumanOverride, |r, p| r.proof = p)?
        } else {
            crate::sign_record(rec, restorer_kp, Record::HumanOverride, |r, p| r.proof = p)?
        };
        let out = Record::HumanOverride(signed);
        self.push(out.clone())?;
        Ok(out)
    }

    pub fn declare_succession(
        &mut self,
        outgoing_kp: &Keypair,
        incoming_kp: &Keypair,
        shard_hash: &str,
        reason: &str,
    ) -> Result<Record> {
        let outgoing_id = outgoing_kp.did();
        let incoming_id = incoming_kp.did();
        let Some(m) = self.members.get(&outgoing_id) else {
            return Err(Error::Policy("outgoing is not a member".into()));
        };
        if tier_rank(&m.tier).unwrap_or(0) < tier_rank("control").unwrap() {
            return Err(Error::Policy("outgoing must be control or higher".into()));
        }
        if outgoing_id == incoming_id {
            return Err(Error::Policy("outgoing and incoming must differ".into()));
        }
        let rec = ModeratorSuccession::new(
            self.world_id.clone(), self.next_seq(), self.last_hash(),
            outgoing_id, incoming_id, shard_hash.into(), reason.into(),
        );
        let signed = crate::sign_record_triple(
            rec, outgoing_kp, incoming_kp, &self.keypair,
            Record::ModeratorSuccession, |r, p| r.proof = p,
        )?;
        let out = Record::ModeratorSuccession(signed);
        self.push(out.clone())?;
        Ok(out)
    }

    pub fn visible_log(&self) -> Vec<serde_json::Value> {
        use serde_json::json;

        let hashes: Vec<String> = self.records.iter()
            .map(|r| r.hash().unwrap_or_default())
            .collect();
        let mut hash_to_idx: HashMap<String, usize> = HashMap::new();
        for (i, h) in hashes.iter().enumerate() {
            hash_to_idx.insert(h.clone(), i);
        }

        let mut redactions_by_target: HashMap<usize, Vec<usize>> = HashMap::new();
        let mut resolved_redacts: HashSet<usize> = HashSet::new();
        for (i, rec) in self.records.iter().enumerate() {
            match rec {
                Record::Redact(r) => {
                    if let Some(&target_idx) = hash_to_idx.get(&r.target_hash) {
                        redactions_by_target.entry(target_idx).or_default().push(i);
                    }
                }
                Record::HumanOverride(o) if o.action == "restore" => {
                    if let Some(&redact_idx) = hash_to_idx.get(&o.overrides_hash) {
                        resolved_redacts.insert(redact_idx);
                    }
                }
                _ => {}
            }
        }

        let mut out = Vec::new();
        for (i, rec) in self.records.iter().enumerate() {
            match rec {
                Record::Message(m) => {
                    let mut still_redacted: Option<usize> = None;
                    let mut restored: Vec<usize> = Vec::new();
                    if let Some(redact_idxs) = redactions_by_target.get(&i) {
                        for &ri in redact_idxs {
                            if resolved_redacts.contains(&ri) {
                                restored.push(ri);
                            } else if still_redacted.is_none() {
                                still_redacted = Some(ri);
                            }
                        }
                    }
                    if let Some(ri) = still_redacted {
                        if let Record::Redact(r) = &self.records[ri] {
                            out.push(json!({
                                "seq": m.seq, "type": "message", "kind": m.kind,
                                "author": m.author_id.as_str(),
                                "text": "[removed]",
                                "redacted": true,
                                "reason": r.reason,
                                "rule": r.rule,
                                "incident": short_hash(&hashes[ri]),
                                "ts": m.ts,
                            }));
                        }
                    } else {
                        let mut obj = json!({
                            "seq": m.seq, "type": "message", "kind": m.kind,
                            "author": m.author_id.as_str(),
                            "text": m.text,
                            "redacted": false,
                            "ts": m.ts,
                        });
                        if !restored.is_empty() {
                            let notes: Vec<serde_json::Value> = restored.iter().map(|&ri| {
                                if let Record::Redact(r) = &self.records[ri] {
                                    json!({"incident": short_hash(&hashes[ri]), "reason": r.reason, "rule": r.rule})
                                } else { json!({}) }
                            }).collect();
                            obj["restored_incidents"] = json!(notes);
                        }
                        out.push(obj);
                    }
                }
                Record::Redact(r) => {
                    out.push(json!({"seq": r.seq, "type": "redact", "by": r.subject_id.as_str(), "reason": r.reason, "target": r.target_hash, "rule": r.rule, "actor_role": r.actor_role, "incident": short_hash(&hashes[i])}));
                }
                Record::Join(j) => {
                    out.push(json!({"seq": j.seq, "type": "join", "subject": j.subject_id.as_str(), "tier": j.tier, "is_ai": j.is_ai}));
                }
                Record::Leave(l) => {
                    out.push(json!({"seq": l.seq, "type": "leave", "subject": l.subject_id.as_str()}));
                }
                Record::HumanOverride(o) => {
                    out.push(json!({"seq": o.seq, "type": "human_override", "by": o.overridden_by.as_str(), "overrides": o.overrides_hash, "action": o.action, "reason": o.reason}));
                }
                Record::Refusal(r) => {
                    out.push(json!({"seq": r.seq, "type": "refused", "requester": r.requester_id.as_str(), "reason": r.reason}));
                }
                Record::RoomCreated(_) => {
                    out.push(json!({"seq": 0, "type": "created"}));
                }
                Record::ModeratorSuccession(s) => {
                    out.push(json!({
                        "seq": s.seq, "type": "moderator_succession",
                        "outgoing": s.outgoing_id.as_str(),
                        "incoming": s.incoming_id.as_str(),
                        "reason": s.reason,
                    }));
                }
                Record::KnowledgeShardExport(_) | Record::PortLanding(_) | Record::Bookmark(_) | Record::Presence(_) => {}
            }
        }
        out
    }
}

pub fn verify_log(log: &RoomLog) -> Result<()> {
    let (world_id, creator_id, template, per_record_mode) = match log.records.first() {
        Some(Record::RoomCreated(rc)) => {
            if rc.world_id != log.world_id {
                return Err(Error::Verification(
                    "room_created world_id does not match log world_id".into(),
                ));
            }
            if rc.template.hash()? != log.template.hash()? {
                return Err(Error::Verification(
                    "room_created template does not match log template".into(),
                ));
            }
            (
                &rc.world_id,
                rc.creator_id.clone(),
                &rc.template,
                rc.template.head_interval <= 1,
            )
        }
        _ => return Err(Error::Chain("log must start with room_created".into())),
    };

    let mut prev: Option<String> = None;
    let mut members: HashMap<DidKey, Member> = HashMap::new();
    let mut redacted: HashSet<String> = HashSet::new();

    for (i, rec) in log.records.iter().enumerate() {
        if rec.seq() != i as u64 {
            return Err(Error::Chain(format!(
                "record {i}: seq {} != expected {}",
                rec.seq(),
                i
            )));
        }
        match rec {
            Record::RoomCreated(_) => {
                if i != 0 {
                    return Err(Error::Chain(format!("record {i}: room_created not at start")));
                }
            }
            _ => {
                if rec.prev() != prev.as_ref() {
                    return Err(Error::Chain(format!("record {i}: prev mismatch")));
                }
            }
        }

        if rec.world_id() != world_id {
            return Err(Error::Verification(format!(
                "record {i}: world_id mismatch"
            )));
        }

        let proofs = rec.proof();

        let requires_room_proof = match rec {
            Record::RoomCreated(_) => true,
            Record::Refusal(_) => true,
            Record::KnowledgeShardExport(_) => true,
            Record::ModeratorSuccession(_) => true,
            Record::Leave(_) => false,
            _ => per_record_mode,
        };
        if requires_room_proof {
            let rp = proofs.find_by_method(world_id).ok_or_else(|| {
                Error::Verification(format!("record {i}: missing room proof"))
            })?;
            rp.verify(&body)?;
        }

        match rec {
            Record::RoomCreated(r) => {
                let creator_proof = proofs.find_by_method(&r.creator_id)
                    .ok_or_else(|| Error::Verification(format!("record {i}: missing creator proof")))?;
                creator_proof.verify(&body)?;
                members.insert(r.creator_id.clone(), Member { tier: "terminal".into(), is_ai: false });
            }
            Record::Message(m) => {
                let Some(mem) = members.get(&m.author_id) else {
                    return Err(Error::Policy(format!("record {i}: author not a member")));
                };
                if tier_rank(&mem.tier).unwrap_or(0) < tier_rank("communicate").unwrap() {
                    return Err(Error::Policy(format!("record {i}: author tier insufficient")));
                }
                if m.text.len() as u64 > template.max_msg_len {
                    return Err(Error::Policy(format!("record {i}: message_too_long")));
                }
                // Rate limit replay
                let window = template.rate_limit_window as usize;
                let cap = template.rate_limit_count as usize;
                let recent = log.records[..i]
                    .iter()
                    .rev()
                    .take(window)
                    .filter(|r| matches!(r, Record::Message(prev_m) if prev_m.author_id == m.author_id))
                    .count();
                if recent >= cap {
                    return Err(Error::Policy(format!("record {i}: rate_limit exceeded")));
                }
                // Inter-AI pace replay
                if mem.is_ai && template.inter_ai_pace > 0 {
                    let mut last_ai_idx: Option<usize> = None;
                    for (prev_idx, prev_rec) in log.records[..i].iter().enumerate().rev() {
                        if let Record::Message(prev_m) = prev_rec {

                                .get(&prev_m.author_id)
                                .map(|x| x.is_ai)

                            if prev_ai {
                                last_ai_idx = Some(prev_idx);
                                break;
                            }
                        }

                    if let Some(idx) = last_ai_idx {
                        let since = (i - idx - 1) as u64;
                        if since < template.inter_ai_pace {
                            return Err(Error::Policy(format!("record {i}: inter_ai_pace violated")));
                        }
                    }
                }
                let author_proof = proofs.find_by_method(&m.author_id)
                    .ok_or_else(|| Error::Verification(format!("record {i}: missing author proof")))?;
                author_proof.verify(&body)?;

                };
                if mem.is_ai {
                    return Err(Error::Policy(format!("record {i}: only humans can override")));
                }
                if tier_rank(&mem.tier).unwrap_or(0) < tier_rank("control").unwrap() {
                    return Err(Error::Policy(format!("record {i}: overrider tier insufficient")));

                };

                        };

                            return Err(Error::Policy(format!("record {i}: override 'none' target must be AI-authored")));
                        }
                    }
                    "restore" => {

                    }
                    other => {

                }
            }

                }

                    .ok_or_else(|| Error::Verification(format!("record {i}: missing incoming proof")))?;
                incoming_proof.verify(&body)?;
                let Some(mem) = members.get(&s.outgoing_id) else {
                    return Err(Error::Policy(format!("record {i}: outgoing not a member")));
                };
                if tier_rank(&mem.tier).unwrap_or(0) < tier_rank("control").unwrap() {
                    return Err(Error::Policy(format!("record {i}: outgoing tier insufficient")));
                }
                if s.outgoing_id == s.incoming_id {
                    return Err(Error::Policy(format!("record {i}: outgoing == incoming")));
                }
                if !looks_like_hash(&s.knowledge_shard_hash) {
                    return Err(Error::Policy(format!("record {i}: knowledge_shard_hash not a 64-hex string")));
                }
            }
        }

        prev = Some(rec.hash()?);
    }

    // Heads
    let mut last_length: u64 = 0;
    for (i, head) in log.heads.iter().enumerate() {
        log.verify_head(head).map_err(|e| Error::Verification(format!("head {i}: {e}")))?;
        if head.length <= last_length {
            return Err(Error::Verification(format!("head {i}: length not increasing")));
        }
        last_length = head.length;
    }
    if per_record_mode && !log.heads.is_empty() {
        return Err(Error::Verification("per-record mode should not emit heads".into()));
    }
    if !per_record_mode {
        let n = template.head_interval;
        let appends = log.records.len().saturating_sub(1) as u64;
        if appends > 0 {
            let periodic = appends / n;
            let rem = appends % n;
            let expected = if rem == 0 { periodic } else { periodic + 1 };
            if log.heads.len() as u64 != expected {
                return Err(Error::Verification(format!(
                    "head count: expected {} for {} appends at interval {}, got {}",
                    expected, appends, n, log.heads.len()
                )));
            }
            for (k, head) in log.heads.iter().enumerate() {
                let k_u64 = k as u64;
                let expected_len = if k_u64 < periodic {
                    1 + (k_u64 + 1) * n
                } else {
                    log.records.len() as u64
                };
                if head.length != expected_len {
                    return Err(Error::Verification(format!(
                        "head {k}: expected length {expected_len}, got {}",
                        head.length
                    )));
                }
            }
        } else if !log.heads.is_empty() {
            return Err(Error::Verification(
                "head mode: no heads expected when no records appended after seed".into(),
            ));
        }
    }

    if template.personal {
        for rec in &log.records {
            if let Record::Join(j) = rec {
                if j.subject_id != creator_id {
                    return Err(Error::Policy("personal room: non-creator joined".into()));
                }
            }
        }
    }

    Ok(())
}

pub fn verify_chain(room: &Room) -> Result<()> {
    verify_log_chain(&room.log())
}

pub fn verify_log_chain(log: &RoomLog) -> Result<()> {
    let mut prev: Option<String> = None;
    for (i, rec) in log.records.iter().enumerate() {
        match rec {
            Record::RoomCreated(_) if i != 0 => {
                return Err(Error::Chain("room_created not at start".into()));
            }
            Record::RoomCreated(_) => {}
            _ => {
                if rec.prev() != prev.as_ref() {
                    return Err(Error::Chain(format!("record {i}: prev mismatch")));
                }
            }
        }
        prev = Some(rec.hash()?);
    }
    Ok(())
}

        .proof
        .find_by_method(&t.porter_id)
        .ok_or_else(|| Error::Verification("missing porter proof".into()))?;
    proof.verify(&body)

                .find(|r| {
                    r.hash().ok().as_deref() == Some(draft_hash)
                        && matches!(r, Record::Message(m) if m.kind == "draft")

                .ok_or_else(|| Error::Policy("draft not found".into()))?;
            match draft { Record::Message(m) => m.text.clone(), _ => unreachable!() }

        let full_text = format!("{prefix}{draft_text}");

        &self,
        port_hash: &str,
        room: &mut Room,
    ) -> Result<Record> {
        let (label, entries_json, target_ok): (String, String, Option<DidKey>) = {
            let port = self.room.records().iter()
                .find(|r| {
                    r.hash().ok().as_deref() == Some(port_hash)
                        && matches!(r, Record::PortLanding(_))

                        .map_err(|e| Error::Serialization(e.to_string()))?;

            if target != room.world_id() {
                return Err(Error::Policy("port target_room does not match this room".into()));
            }
        }
        let prefix = format!("[home_ref={}] [ported from {}] ", &port_hash[..16.min(port_hash.len())], label);
        let full_text = format!("{prefix}{entries_json}");
        room.post(&self.user_kp, &full_text, "port")
    }

        for rec in self.room.records() {
            let body = serde_json::to_value(rec)
                .map_err(|e| Error::Serialization(e.to_string()))?;
            records.push(serde_json::json!({"hash": rec.hash()?, "body": body}));
        }
        Ok(serde_json::json!({
            "world_id": self.room.world_id().as_str(),
            "template": self.room.template(),
            "records": records,
        }))
    }

pub fn verify_commons(commons: &Commons) -> Result<()> {
    let mut prev: Option<String> = None;
    for (i, rec) in commons.records().iter().enumerate() {
        match rec {
            CommonsRecord::RoomRegistered(r) => {
                if r.commons_id != commons.world_id {
                    return Err(Error::Verification(format!("record {i}: commons_id mismatch")));

                if r.prev != prev {
                    return Err(Error::Chain(format!("record {i}: prev mismatch")));
                }
                let body = hash::body_for_signing(r)?;
                let reg = r.proof.find_by_method(&r.registered_by)
                    .ok_or_else(|| Error::Verification(format!("record {i}: missing registrant proof")))?;
                reg.verify(&body)?;
                let com = r.proof.find_by_method(&commons.world_id)
                    .ok_or_else(|| Error::Verification(format!("record {i}: missing commons proof")))?;
                com.verify(&body)?;
            }
            CommonsRecord::SiblingDeclared(s) => {
                if s.commons_id != commons.world_id {
                    return Err(Error::Verification(format!("record {i}: commons_id mismatch")));
                }
                if s.prev != prev {
                    return Err(Error::Chain(format!("record {i}: prev mismatch")));
                }
                if s.parent_id == s.sibling_id {
                    return Err(Error::Policy(format!("record {i}: parent == sibling")));
                }
                if s.shard_hash.len() != 64 || !s.shard_hash.chars().all(|c| c.is_ascii_hexdigit()) {
                    return Err(Error::Policy(format!("record {i}: bad shard_hash")));
                }
                let body = hash::body_for_signing(s)?;
                let p = s.proof.find_by_method(&s.parent_id)

                let sib = s.proof.find_by_method(&s.sibling_id)

                let com = s.proof.find_by_method(&commons.world_id)
                    .ok_or_else(|| Error::Verification(format!("record {i}: missing commons proof")))?;
