    use chat_world::hash::body_for_signing;
    use chat_world::record::*;
    use chat_world::{Proof, ProofOrArray};

    fn sign_single(rec: &Record, signer: &Keypair) -> ProofOrArray {
        let body = body_for_signing(rec).unwrap();
        let sig = signer.sign(&body);
        ProofOrArray::Single(Proof::create(signer, &sig))
    }

    fn sign_dual(rec: &Record, subj: &Keypair, room_kp: &Keypair)
        -> ProofOrArray {
        let body = body_for_signing(rec).unwrap();
        let s1 = subj.sign(&body);
        let s2 = room_kp.sign(&body);
        ProofOrArray::array(vec![
            Proof::create(subj, &s1),
            Proof::create(room_kp, &s2),
        ])
    }

Seven facts about the crate as it stands:

  - Room::create takes the room's keypair by value. room.keypair()
    gives it back by reference, so a test that makes a room can sign
    as the room.

    well (head_interval is 1). A message there carries two proofs:
    its author's and the room's.
  - post gives back the record the room wrote: a message, or a
    refusal when the room refused it, Ok either way. A room made by
    RoomTemplate::chat refuses a message from an author who already
    wrote ten of its last twenty records.
  - Room::create makes its maker a member who may remove and
    restore. Who else may is in src/room.rs.
  - draw_at with no heads is draw.

    refuses an open room whose law hides its count, and gives its
